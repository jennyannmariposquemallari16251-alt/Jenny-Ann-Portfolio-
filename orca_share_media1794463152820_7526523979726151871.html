This is the Java console version of my Guess the Word game. It asks the player to choose a subject and difficulty, gives a clue, and checks the player's answer.

import java.util.Scanner;
import java.util.Random;

public class GuessTheWord {
    static Scanner input = new Scanner(System.in);
    static Random random = new Random();

    public static void main(String[] args) {
        String[] subjects = {"Math", "Science", "English"};
        String[][] easyWords = {
            {"sum", "Result of addition"},
            {"sun", "The star that gives Earth light"},
            {"noun", "A word that names a person, place, or thing"}
        };

        System.out.println("=== GUESS THE WORD ===");
        System.out.println("1. Math  2. Science  3. English");
        System.out.print("Choose a subject: ");
        int subjectChoice = input.nextInt();
        input.nextLine();

        if (subjectChoice < 1 || subjectChoice > 3) {
            System.out.println("Invalid subject choice.");
            return;
        }

        System.out.println("Choose difficulty: 1. Easy  2. Medium  3. Hard");
        System.out.print("Enter difficulty: ");
        int difficulty = input.nextInt();
        input.nextLine();

        if (difficulty < 1 || difficulty > 3) {
            System.out.println("Invalid difficulty choice.");
            return;
        }

        String[][] words = getWords(subjectChoice - 1, difficulty);
        int chosen = random.nextInt(words.length);
        String answer = words[chosen][0];
        String clue = words[chosen][1];
        int attempts = 3;

        System.out.println("Clue: " + clue);

        while (attempts > 0) {
            System.out.print("Your guess: ");
            String guess = input.nextLine().trim();

            if (guess.equalsIgnoreCase(answer)) {
                System.out.println("Correct! Good job!");
                break;
            } else {
                attempts--;
                if (attempts > 0) {
                    System.out.println("Try again. Attempts left: " + attempts);
                } else {
                    System.out.println("No more attempts. The answer is: " + answer);
                }
            }
        }
    }

    static String[][] getWords(int subject, int difficulty) {
        String[][][] data = {
            {
                {"sum", "Result of addition"},
                {"even", "A number divisible by two"},
                {"fraction", "A part of a whole"}
            },
            {
                {"sun", "The star that gives Earth light"},
                {"oxygen", "Gas humans need to breathe"},
                {"gravity", "Force that pulls objects toward Earth"}
            },
            {
                {"noun", "A word that names a person, place, or thing"},
                {"adjective", "A word that describes a noun"},
                {"synonym", "A word with a similar meaning"}
            }
        };

        // The sample console game uses the same word set for each level.
        return data[subject];
    }
}
